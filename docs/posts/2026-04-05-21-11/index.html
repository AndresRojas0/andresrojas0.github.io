/* ── Cart state (persisted to localStorage) ─────────────── */
const CART_KEY = 'wickes_spa_cart';
function loadCart() {
  try { return JSON.parse(localStorage.getItem(CART_KEY)) || {}; }
  catch (e) { return {}; }
}
function saveCart(cart) { localStorage.setItem(CART_KEY, JSON.stringify(cart)); }
let CART = loadCart();

function cartLines() {
  return Object.entries(CART)
    .filter(([sku, qty]) => qty > 0 && PRODUCTS_BY_SKU[sku])
    .map(([sku, qty]) => {
      const p = PRODUCTS_BY_SKU[sku];
      return { sku, qty, product: p, lineTotal: p.price * qty };
    });
}
function cartCount() { return cartLines().reduce((s, l) => s + l.qty, 0); }
function cartSubtotal() { return cartLines().reduce((s, l) => s + l.lineTotal, 0); }

function addToCart(sku, qty = 1) {
  CART[sku] = (CART[sku] || 0) + qty;
  saveCart(CART);
  updateCartBadge();
}
function setCartQty(sku, qty) {
  qty = Math.max(1, Math.min(99, qty || 1));
  CART[sku] = qty;
  saveCart(CART);
  updateCartBadge();
}
function removeFromCart(sku) {
  delete CART[sku];
  saveCart(CART);
  updateCartBadge();
}
function clearCartAll() {
  CART = {};
  saveCart(CART);
  updateCartBadge();
}
function updateCartBadge() {
  const badge = document.getElementById('cart-badge');
  if (badge) {
    const n = cartCount();
    badge.textContent = n;
    badge.setAttribute('aria-label', n + ' items in cart');
  }
}

function handleAddToCart(sku, btn, qty = 1) {
  const p = PRODUCTS_BY_SKU[sku];

/* ════════════════════════════════════════════════════════════
   ROUTER
════════════════════════════════════════════════════════════ */
function parseHash() {
  let hash = location.hash || '#/';
  hash = hash.slice(1); // drop '#'
  if (!hash.startsWith('/')) hash = '/' + hash;
  const [path, qs] = hash.split('?');
  return { path: path.replace(/\/+$/, '') || '/', params: new URLSearchParams(qs || '') };
}

function navigateHash(hash) {
  if (location.hash === hash) { router(); } else { location.hash = hash; }
}

function router() {
  const { path, params } = parseHash();
  const app = document.getElementById('app');
  app.classList.remove('app-fade');

  updateCartBadge();
  highlightNav(path, params);

  if (path === '/catalog') {
    renderCatalog(params);
  } else if (path === '/product') {
    renderProduct(params.get('sku'));
  } else if (path === '/cart') {
    renderCart();
  } else {
    renderHome();
  }

  window.scrollTo(0, 0);
  requestAnimationFrame(() => app.classList.add('app-fade'));
}

function highlightNav(path, params) {
  const cat = params.get('cat');
  document.querySelectorAll('#main-nav [data-cat-link]').forEach(a => {
    a.classList.toggle('route-active', path === '/catalog' && a.dataset.catLink === cat);
  });
  const searchInput = document.getElementById('search-input');
  if (searchInput) searchInput.value = path === '/catalog' ? (params.get('q') || '') : searchInput.value;
}

window.addEventListener('hashchange', router);

document.addEventListener('click', (e) => {
  const link = e.target.closest('[data-route-link]');
  if (link) {
    // allow default hash navigation, but make sure router runs even if hash unchanged
    const href = link.getAttribute('href') || '';
    if (href.startsWith('#')) {
      setTimeout(router, 0);
    }
  }
});

document.getElementById('search-form').addEventListener('submit', (e) => {
  e.preventDefault();
  const q = document.getElementById('search-input').value.trim();
  navigateHash('#/catalog' + (q ? '?q=' + encodeURIComponent(q) : ''));
});

    cats: cat ? new Set([cat]) : new Set(),
    q,
    min: null, max: null,
    sort: 'featured',

    if (state.cats.size > 0) {
      let ok = false;
      state.cats.forEach(c => { if (p.cat.includes(c)) ok = true; });
      if (!ok) return false;
    }
    if (state.q) {
      const hay = (p.name + ' ' + p.desc + ' ' + p.sku).toLowerCase();
      if (!hay.includes(state.q)) return false;
    }
    if (state.min != null && p.price < state.min) return false;
    if (state.max != null && p.price > state.max) return false;
    return true;
  }

  function renderGrid() {
    let list = PRODUCTS.filter(matches);
    if (state.sort === 'price-asc') list = list.slice().sort((a, b) => a.price - b.price);
    else if (state.sort === 'price-desc') list = list.slice().sort((a, b) => b.price - a.price);
    else if (state.sort === 'name-asc') list = list.slice().sort((a, b) => a.name.localeCompare(b.name));

    const grid = document.getElementById('product-grid');
    grid.innerHTML = list.length
      ? list.map(productCardHTML).join('')

      renderGrid();
    });
  });

  document.getElementById('sort-select').addEventListener('change', (e) => {
    state.sort = e.target.value; renderGrid();
  });

  document.getElementById('apply-price-btn').addEventListener('click', () => {
    const minV = parseFloat(document.getElementById('price-min').value);
    const maxV = parseFloat(document.getElementById('price-max').value);
    state.min = isNaN(minV) ? null : minV;
    state.max = isNaN(maxV) ? null : maxV;
    renderGrid();
    showToast('Filtered: showing ' + PRODUCTS.filter(matches).length + ' products');
  });

  document.getElementById('clear-filters-btn').addEventListener('click', () => {
    state = { cats: new Set(), q: '', min: null, max: null, sort: 'featured', view: state.view };

    renderGrid();
    showToast('All filters cleared');
  });

  document.getElementById('view-grid').addEventListener('click', (e) => {
    state.view = 'grid';

    renderGrid();
  });

    state.view = 'list';
    e.currentTarget.style.background = 'var(--blue)'; e.currentTarget.style.color = 'var(--yellow)'; e.currentTarget.style.borderColor='var(--blue)';

    g.style.background = 'var(--grey-light)'; g.style.color = 'var(--blue-dark)'; g.style.borderColor='var(--grey)';
    renderGrid();
  });

  document.querySelectorAll('.sidebar__title[role="button"]').forEach(title => {
    title.addEventListener('click', () => {
      const body = title.nextElementSibling;
      const isOpen = title.classList.contains('open');
      title.classList.toggle('open', !isOpen);
      title.setAttribute('aria-expanded', String(!isOpen));
      if (body) body.style.display = isOpen ? 'none' : '';
    });
  });

  renderGrid();
  CURRENT_RENDER_REFRESH = renderGrid;
}

    const discount = promoApplied ? subtotal * 0.2 : 0;
    const total = subtotal - discount;

    document.getElementById('subtotal').textContent = fmt(subtotal);
    document.getElementById('vat-amount').textContent = fmt(vat);
    document.getElementById('item-count').textContent = count;
    document.getElementById('grand-total').textContent = fmt(total);

    const savingsRow = document.getElementById('savings-row');
    if (discount > 0) {
